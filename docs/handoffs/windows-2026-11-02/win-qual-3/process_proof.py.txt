    handle=k.OpenProcess(0x1000|0x00100000,False,row['pid'])
    if not handle:return None
    created,exited,kernel,user=w.FILETIME(),w.FILETIME(),w.FILETIME(),w.FILETIME()
    assert k.GetProcessTimes(handle,ctypes.byref(created),ctypes.byref(exited),ctypes.byref(kernel),ctypes.byref(user))
    assert (created.dwHighDateTime<<32)|created.dwLowDateTime==row['birth_filetime']
