def audio(bundle: Path, name='recording.wav'):
    bundle.mkdir(parents=True, exist_ok=True)
    with wave.open(str(bundle / name), 'wb') as stream:
        stream.setparams((1, 2, 48000, 0, 'NONE', 'not compressed'))
        stream.writeframes(bytes(12 * 96000))

def transcript(bundle: Path, label='20261031-170000', cleanup=False):
    path = write_published_run(bundle / 'transcripts' / label, deletion={'confirmed': not cleanup})
    data = json.loads(path.read_text(encoding='utf-8'))
    data['speaker_names'] = {'SPEAKER_00': 'Alex <draft>', 'SPEAKER_01': 'Alex <draft>'}
    segments = [(8,10,'Later source passage appears first. [a.*] literal.', 'SPEAKER_00'),
                (0,2,'Opening <img onerror=bad()> [a.*] literal.', 'SPEAKER_01'),
                (1,3,'Overlapping passage about recording.', 'SPEAKER_00'),
                (5,5,'Zero duration remains readable.', 'SPEAKER_00'),
                (25,30,'Beyond the available recording.', 'SPEAKER_01')]
    segments += [(10+i*.04,10.04+i*.04,f'Synthetic review passage {i+6}: a longer sentence for scrolling and filtering.', 'SPEAKER_0'+str(i%2)) for i in range(25)]
    data['segments'] = [{'id':7,'start':start,'end':end,'text':text,'speaker':speaker,'words':[{'word':text,'start':start,'end':end,'probability':1,'speaker':speaker}]} for start,end,text,speaker in segments]
    data['text'] = ' '.join(segment['text'] for segment in data['segments'])
    parse_canonical(data)
    path.write_text(json.dumps(data), encoding='utf-8')

def hashes():
    import hashlib
    return {str(path.relative_to(fixture)):hashlib.sha256(path.read_bytes()).hexdigest() for path in fixture.rglob('*') if path.is_file()}
